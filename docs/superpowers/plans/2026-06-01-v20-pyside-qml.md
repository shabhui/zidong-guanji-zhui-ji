# AutoShutdown v2.0-preview PySide6/QML Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build AutoShutdown v2.0-preview with PySide6 + QML Fluent Neon UI, covering countdown, fixed time, power actions, dry-run safety, and a visually impressive Control Deck.

**Architecture:** Python + PySide6 provides QML engine. One `AppController` QObject exposes app state/properties/slots to QML. `Main.qml` defines the entire single-window UI. `Theme.qml` holds semantic colors/spacing/animation tokens. Power actions use `ctypes` to call Windows API; `dryRun` flag gates them in dev.

**Tech Stack:** Python 3.12, PySide6 6.11, Qt Quick Controls (FluentWinUI3 fallback to custom Neumorphism via Theme), `ctypes` for power, `QTimer` for countdown.

```python
import sys
from pathlib import Path
from PySide6.QtGui import QGuiApplication
from PySide6.QtQml import QQmlApplicationEngine
from PySide6.QtQuickControls2 import QQuickStyle
from controller import AppController

if __name__ == "__main__":
    app = QGuiApplication(sys.argv)
    app.setApplicationName("AutoShutdownQt")
    app.setApplicationVersion("2.0-preview")

    # Try FluentWinUI3; fall back gracefully
    available = QQuickStyle.availableStyles()
    style_names = [s.lower() for s in available]
    if "fluentwinui3" in style_names:
        QQuickStyle.setStyle("FluentWinUI3")
    elif "fusion" in style_names:
        QQuickStyle.setStyle("Fusion")

    controller = AppController()

    engine = QQmlApplicationEngine()
    context = engine.rootContext()
    context.setContextProperty("controller", controller)

    qml_dir = Path(__file__).parent / "qml"
    engine.addImportPath(str(qml_dir))

    main_qml = qml_dir / "Main.qml"
    engine.load(str(main_qml))

    if not engine.rootObjects():
        sys.exit(-1)

    sys.exit(app.exec())
```

Write `AutoShutdownQt/controller.py`:

```python
from PySide6.QtCore import QObject, Signal, Property, Slot
from PySide6.QtCore import QTimer
from datetime import datetime, timedelta


class AppController(QObject):
    # Signals
    remainingTimeChanged = Signal()
    statusChanged = Signal()
    targetInfoChanged = Signal()
    dryRunChanged = Signal()

    POWER_ACTIONS = ["shutdown", "sleep", "hibernate", "restart", "logoff", "lock"]

    def __init__(self, parent=None):
        super().__init__(parent)
        self._dry_run = True
        self._selected_action = "shutdown"
        self._status = "ready"  # ready, running, paused
        self._remaining_seconds = 0
        self._target_time_str = ""
        self._force_close = False
        self._timer = QTimer(self)
        self._timer.setInterval(1000)
        self._timer.timeout.connect(self._on_tick)

    # --- Properties exposed to QML ---

    def getDryRun(self):
        return self._dry_run

    def setDryRun(self, val):
        if self._dry_run != val:
            self._dry_run = val
            self.dryRunChanged.emit()

    dryRun = Property(bool, getDryRun, setDryRun, notify=dryRunChanged)

    def getStatus(self):
        return self._status

    status = Property(str, getStatus, notify=statusChanged)

    def getRemainingText(self):
        h = self._remaining_seconds // 3600
        m = (self._remaining_seconds % 3600) // 60
        s = self._remaining_seconds % 60
        return f"{h:02d}:{m:02d}:{s:02d}"

    remainingText = Property(str, getRemainingText, notify=remainingTimeChanged)

    def getTargetInfo(self):
        return self._target_time_str

    targetInfo = Property(str, getTargetInfo, notify=targetInfoChanged)

    def getSelectedAction(self):
        return self._selected_action

    def setSelectedAction(self, val):
        if val in self.POWER_ACTIONS and self._selected_action != val:
            self._selected_action = val
            self.targetInfoChanged.emit()

    selectedAction = Property(str, getSelectedAction, setSelectedAction, notify=targetInfoChanged)

    def getForceClose(self):
        return self._force_close

    def setForceClose(self, val):
        if self._force_close != val:
            self._force_close = val

    forceClose = Property(bool, getForceClose, setForceClose)

    def getActionLabel(self):
        labels = {
            "shutdown": "关机",
            "sleep": "睡眠",
            "hibernate": "休眠",
            "restart": "重启",
            "logoff": "注销",
            "lock": "锁定",
        }
        return labels.get(self._selected_action, "")

    actionLabel = Property(str, getActionLabel, notify=targetInfoChanged)

    # --- Slots ---

    @Slot(int, int, int)
    def startCountdown(self, hours, minutes, seconds):
        total = hours * 3600 + minutes * 60 + seconds
        if total <= 0:
            return
        self._remaining_seconds = total
        self._status = "running"
        self._target_time_str = ""
        self._timer.start()
        self.statusChanged.emit()
        self.remainingTimeChanged.emit()

    @Slot(int, int)
    def startFixedTime(self, hour, minute):
        now = datetime.now()
        target = datetime(now.year, now.month, now.day, hour, minute, 0)
        if target <= now:
            target += timedelta(days=1)
        delta = int((target - now).total_seconds())
        if delta <= 0:
            return
        self._remaining_seconds = delta
        self._status = "running"
        self._target_time_str = target.strftime("%Y-%m-%d %H:%M")
        self._timer.start()
        self.statusChanged.emit()
        self.remainingTimeChanged.emit()
        self.targetInfoChanged.emit()

    @Slot()
    def cancel(self):
        self._timer.stop()
        self._remaining_seconds = 0
        self._status = "ready"
        self._target_time_str = ""
        self.statusChanged.emit()
        self.remainingTimeChanged.emit()
        self.targetInfoChanged.emit()

    @Slot()
    def executeNow(self):
        if self._dry_run:
            print(f"[dryRun] Would execute: {self._selected_action} force={self._force_close}")
            return
        self._execute_power_action()

    def _on_tick(self):
        self._remaining_seconds -= 1
        self.remainingTimeChanged.emit()
        if self._remaining_seconds <= 0:
            self._timer.stop()
            self._status = "ready"
            self.statusChanged.emit()
            if not self._dry_run:
                self._execute_power_action()
            else:
                print(f"[dryRun] Would have executed: {self._selected_action}")

    def _execute_power_action(self):
        from power_service import execute_power_action
        execute_power_action(self._selected_action, self._force_close)
```

- [ ] **Step 1: Create power_service.py**

Write `AutoShutdownQt/power_service.py`:

```python
import subprocess
import ctypes

# Token privileges for shutdown
TOKEN_ADJUST_PRIVILEGES = 0x0020
TOKEN_QUERY = 0x0008
SE_SHUTDOWN_NAME = "SeShutdownPrivilege"
SE_PRIVILEGE_ENABLED = 0x00000002
EWX_SHUTDOWN = 0x00000001
EWX_REBOOT = 0x00000002
EWX_FORCE = 0x00000004
EWX_FORCEIFHUNG = 0x00000010


def _enable_shutdown_privilege():
    """Enable SeShutdownPrivilege for the current process."""
    import ctypes.wintypes
    advapi32 = ctypes.windll.advapi32
    kernel32 = ctypes.windll.kernel32

    hToken = ctypes.wintypes.HANDLE()
    if not advapi32.OpenProcessToken(
        kernel32.GetCurrentProcess(),
        TOKEN_ADJUST_PRIVILEGES | TOKEN_QUERY,
        ctypes.byref(hToken),
    ):
        return False

    luid = ctypes.wintypes.LARGE_INTEGER()
    if not advapi32.LookupPrivilegeValueW(None, SE_SHUTDOWN_NAME, ctypes.byref(luid)):
        kernel32.CloseHandle(hToken)
        return False

    tp = (luid, SE_PRIVILEGE_ENABLED)
    advapi32.AdjustTokenPrivileges(hToken, False, ctypes.byref(tp), 0, None, None)
    kernel32.CloseHandle(hToken)
    return True


def _exit_windows(flags):
    _enable_shutdown_privilege()
    ctypes.windll.user32.ExitWindowsEx(flags, 0)


def execute_power_action(action: str, force: bool = False):
    flags_map = {
        "shutdown": EWX_SHUTDOWN,
        "restart": EWX_REBOOT,
        "logoff": 0,  # handled separately
    }

    if action == "logoff":
        ctypes.windll.user32.ExitWindowsEx(0, 0)
        return

    if action == "lock":
        ctypes.windll.user32.LockWorkStation()
        return

    if action == "sleep":
        subprocess.run(["rundll32.exe", "powrprof.dll,SetSuspendState", "0", "1", "0"], check=False)
        return

    if action == "hibernate":
        subprocess.run(["rundll32.exe", "powrprof.dll,SetSuspendState", "0", "0", "0"], check=False)
        return

    flags = flags_map.get(action, 0)
    if force and action in ("shutdown", "restart"):
        flags |= EWX_FORCE | EWX_FORCEIFHUNG

    if flags:
        _exit_windows(flags)
```
